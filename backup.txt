from fastapi import FastAPI, UploadFile, File
import numpy as np
import cv2
import face_recognition

app = FastAPI()


@app.get("/")
def home():
    return {"message": "Server aktif!"}


@app.post("/recognize")
async def recognize(file: UploadFile = File(...)):

    image = await file.read()

    array = np.frombuffer(
        image,
        dtype=np.uint8
    )

    image_fix = cv2.imdecode(
        array,
        cv2.IMREAD_COLOR
    )

    frame = image_fix

    resize = cv2.resize(
        frame,
        (0, 0),
        fx=0.25,
        fy=0.25
    )

    grey = cv2.cvtColor(
        resize,
        cv2.COLOR_BGR2RGB
    )

    track = face_recognition.face_locations(grey)

    trainer = "Yuda.jpg"

    img_trainer = face_recognition.load_image_file(
        f"./img/{trainer}"
    )

    encode_trainer = face_recognition.face_encodings(
        img_trainer
    )[0]

    if track == []:
        return {
            "pesan": "Tidak ada wajah"
        }

    img_encode = face_recognition.face_encodings(
        grey,
        track
    )

    for (atas, kanan, bawah, kiri), posisi in zip(
        track,
        img_encode
    ):

        compare = face_recognition.compare_faces(
            [encode_trainer],
            posisi
        )

        if True in compare:
            return {
                "pesan": "Correct!",
                "filename": file.filename
            }

        else:
            return {
                "pesan": "erawrrr"
            }
